FUNKTIONIEN FÜR STATISIKEN
/*
GIB = return
FÜR = for-Schleife
SOLANGE = while-Schleife
WENN = if Verzweigung
SONST = else
FUNKTION (Rückgabetyp) Funktionsname (variable variablenname, ...): = Rückgabetyp Funktionsname(variable variablenname, ...){} 
Im Falle nicht vorhandener Information "Rückgabetyp" -> Umsetzung in C als void
/*Funktion*/ bedeutet, dass die Funktion nicht mehr relevant und korrekt ist. Die Funktion wird drin gelassen falls sie im späteren Verlauf 
doch noch benötigt wird. Somit muss man die Funktion nichtmehr neu schreiben sondern nur aktualisieren.
*/

FUNKTION (SimulationsStats) init_statistik():
    Fließkommazahl *auslastung_pro_zeitschritt = malloc(sizeof(Fließkommazahl))
    WENN auslastung_pro_zeitschritt == NULL:
        Ausgabe: "Fehler bei Speicherreservierung für auslastung_pro_zeitschritt"
    ENDE WENN
    Ganzzahl *warteschlange_pro_zeitschritt = malloc(sizeof(Ganzzahl))
    WENN warteschlange_pro_zeitschritt == NULL:
        Ausgabe: "Fehler bei Speicherreservierung für warteschlange_pro_zeitschritt"
    ENDE WENN
    Ganzzahl *wartezeit_pro_zeitschritt = malloc(sizeof(Ganzzahl))
    WENN wartezeit_pro_zeitschritt == NULL:
        Ausgabe: "Fehler bei Speicherreservierung für wartezeit_pro_zeitschritt"
    ENDE WENN

    Fließkommazahl maximale_auslastung = 0
    Ganzzahl maximale_warteschlangenlaenge = 0
    auslastung_pro_zeitschritt = 0
    warteschlange_pro_zeitschritt = 0
    wartezeit_pro_zeitschritt = 0

    SimulationsStats *statistik = malloc(sizeof(SimulationsStats))
    WENN statistik == NULL
        Ausgabe: "Fehler bei Speicherreservierung für statistik"
    ENDE WENN

    statistik->auslastung_pro_zeitschritt = auslastung_pro_zeitschritt
    statistik->warteschlange_pro_zeitschritt = warteschlange_pro_zeitschritt
    statistik->wartezeit_pro_zeitschritt = wartezeit_pro_zeitschritt
    statistik->maximale_auslastung = maximale_auslastung
    statistik->maximale_warteschlangenlaenge = maximale_warteschlangenlaenge

    GIB statistik
ENDE FUNKTION


FUNKTION (Ganzzahl) berechne_aktuelle_auslastung(Parkhaus *p_parkhaus):
    GIB (p_parkhaus->belegte_parkplaetze/p_parkhaus->anzahl_parkplaetze) * 100
ENDE FUNKTION
