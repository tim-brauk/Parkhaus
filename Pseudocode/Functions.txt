FUNCTION init_Parkhaus (parkhaus pparkhaus, Ganzzahl parkplaetze, Ganzzahl maximale_parkdauer, auto erstes_auto_in_der_warteschlange):
    pparkhaus.parkplaetze = parkplaetze
    pparkhaus.maximale_Parkdauer = maximale_Parkdauer
    pparkhaus.erstes_auto_in_der_warteschlange = erstes_auto_in_der_warteschlange
END FUNCTION

FUNCTION init_Auto (parkhaus pparkhaus, auto aauto, Ganzzahl id, Ganzzahl aktuelle_parkdauer, auto naechstes_Auto):
    aauto.id = id
    aauto.parkdauer = aktuelle_parkdauer
    aauto.restliche_parkdauer = pparkhaus.maximale_Parkdauer - aktuelle_parkdauer
    aauto.naechstes_Auto = naechstes_Auto
END FUNCTION

FUNCTION add_car_queue(parkhaus pparkhaus, auto aauto):
    IF pparkhaus.erstes_auto_in_der_warteschlange == NULL:
        pparkhaus.erstes_auto_in_der_warteschlange = aauto
        END FUNCTION //return
    END IF
    Ganzzahl *p_aktuelles_auto = malloc(sizeof(auto))
    p_aktuelles_auto = pparkhaus.erstes_auto_in_der_warteschlange
    WHILE (p_aktuelles_auto.naechstes_auto != NULL):
        p_aktuelles_auto = p_aktuelles_auto.naechstes_Auto
    END WHILE
    p_aktuelles_auto.naechstes_auto = auto
    free(p_aktuelles_auto)
END FUNCTION
