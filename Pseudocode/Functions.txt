FUNCTIONS

FUNCTION initParkhaus (parkhaus Parkhaus, Ganzzahl parkplaetze, Ganzzahl maximale_Parkdauer, auto erstes_auto_in_der_Warteschlange):
    Parkhaus.parkplaetze = parkplaetze
    Parkhaus.maximale_Parkdauer = maximale_Parkdauer
    Parkhaus.erstes_auto_in_der_Warteschlange = erstes_auto_in_der_Warteschlange
END FUNCTION

FUNCTION initAuto (parkhaus Parkhaus, auto Auto, Ganzzahl id, Ganzzahl aktuelle_parkdauer, auto naechstes_Auto):
    Auto.id = id
    Auto.parkdauer = aktuelle_parkdauer
    Auto.restliche_parkdauer = Parkhaus.maximale_Parkdauer - aktuelle_parkdauer
    Auto.naechstes_Auto = naechstes_Auto
END FUNCTION
