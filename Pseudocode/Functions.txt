FUNKTION (Parkhaus) init_parkhaus (Parkplatz *p_parkplaetze, Ganzzahl maximale_parkdauer):
    Parkhaus *p_parkhaus = malloc(sizeof(Parkhaus))
    p_parkhaus->p_parkplaetze = p_parkplaetze
    p_parkhaus->maximale_parkdauer = maximale_parkdauer
    p_parkhaus->p_erstes_kfz_in_der_warteschlange = NULL
    GIB p_parkhaus
ENDE FUNKTION 

FUNKTION (Kfz) init_kfz (Parkhaus *p_parkhaus, Ganzzahl id):
    Kfz *p_kfz = malloc(sizeof(Kfz))
    p_kfz.id = id
    p_kfz.parkdauer = 0
    p_kfz->p_naechstes_kfz = NULL //wird bei kfz_hinzufügen_warteschlange gesetzt
    GIB p_kfz
ENDE FUNKTION 

FUNKTION kfz_hinzufügen_warteschlange(Parkhaus *p_parkhaus, Kfz *p_kfz):
    WENN p_parkhaus->p_erstes_kfz_in_der_warteschlange == NULL:
        p_parkhaus->p_erstes_kfz_in_der_warteschlange = p_kfz
        p_kfz->p_naechstes_kfz = NULL
        GIB 0
        ENDE FUNKTION
    ENDE WENN
    Kfz *p_aktuelles_kfz = p_parkhaus->p_erstes_kfz_in_der_warteschlange
    SOLANGE (p_aktuelles_kfz->p_naechstes_kfz != NULL):
        p_aktuelles_kfz = p_aktuelles_kfz->p_naechstes_kfz
    ENDE SOLANGE
    p_aktuelles_kfz->p_naechstes_kfz = p_kfz
    p_kfz->p_naechstes_kfz = NULL
    ENDE FUNKTION

FUNKTION (Kfz) entferne_kfz_warteschlange(Parkhaus *p_parkhaus):
    WENN p_parkhaus->p_erstes_kfz_in_der_warteschlange == NULL:
        Ausgabe: "Keine kfzs in der Warteschlange"
        GIB 0
        ENDE FUNKTION
    ENDE WENN
    WENN p_parkhaus->p_erstes_kfz_in_der_warteschlange->naechstes_kfz == NULL:
        Kfz *p_kfz_zwischenspeicher = p_parkhaus->p_erstes_kfz_in_der_warteschlange
        p_parkhaus->p_erstes_kfz_in_der_warteschlange = NULL
        GIB p_kfz_zwischenspeicher
        ENDE FUNKTION 
    ENDE WENN
    Kfz *p_kfz_zwischenspeicher = malloc(sizeof(Kfz))
    p_kfz_zwischenspeicher = p_parkhaus->p_erstes_kfz_in_der_warteschlange->p_naechstes_kfz
    p_parkhaus->p_erstes_kfz_in_der_warteschlange->p_naechstes_kfz = NULL
    p_parkhaus->p_erstes_kfz_in_der_warteschlange = p_kfz_zwischenspeicher
ENDE FUNKTION GIB p_kfz_zwischenspeicher

    ENDE FÜR

FUNKTION enferne_kfz(Parkhaus *p_parkhaus, Kfz *p_kfz):
    FÜR Ganzzahl i = 0, i < (sizeof(p_parkhaus->parkplaetze) / 4), i++: // Besprechung nötig wegen der Liste parkplaetze
        WENN p_kfz == p_parkhaus->parkplaetze[i].Kfz:
            p_parkhaus->parkplaetze[i].belegt = 0
            free(p_kfz)
            p_parkhaus->parkplaetze[i].kfz = NULL
            GIB NULL
            ENDE FUNKTION
        ENDE WENN
    ENDE FÜR

FUNKTION enferne_kfzs_maximale_parkdauer(Parkhaus *p_parkhaus):
    FÜR Ganzahl i = 0, i < (sizeof(p_parkhaus->parkplaetze) / sizeof(parkplatz)), i++:
        WENN p_parkhaus->parkplaetze[i].Kfz != NULL:
            WENN p_parkhaus->parkplaetze[i].Kfz.parkdauer >= p_parkhaus->maximale_parkdauer:
                p_parkhaus->parkplaetze[i].belegt = 0
                free(p_parkhaus->parkplaetze[i]->Kfz)
                p_parkhaus->parkplaetze[i]->Kfz = NULL //Falls Parkdauer überschritten fährt das Kfz raus
            ENDE WENN
        ENDE WENN
    ENDE FÜR
ENDE FUNKTION
