FUNKTION (*Simulationsparameter) init_simulationsparameter (Ganzzahl anzahl_parkplaetze, Ganzzahl maximale_parkdauer, float simulations_dauer, Ganzzahl seed, float wahrscheinlichkeit_neues_kfz, float wahrscheinlichkeit_kfz_rausfahren):
    Simulationsparameter *p_simulationsparameter = malloc(sizeof(Simulationsparameter))
    p_simulationsparameter->anzahl_parkplaetze = anzahl_parkplaetze
    p_simulationsparameter->maximale_parkdauer = maximale_parkdauer
    p_simulationsparameter->simulations_dauer = simulations_dauer
    p_simulationsparameter->seed = seed
    p_simulationsparameter->wahrscheinlichkeit_neues_kfz = wahrscheinlichkeit_neues_kfz
    p_simulationsparameter->wahrscheinlichkeit_kfz_rausfahren = wahrscheinlichkeit_kfz_rausfahren
    GIB p_simulationsparameter
ENDE FUNKTION

FUNKTION (*Parkplatz) init_parkplatz():
    Parkplatz *p_parkplatz = malloc(sizeof(Parkplatz))
    p_parkplatz->belegt = 0
    p_parkplatz->p_kfz = NULL
    GIB p_parkplatz
ENDE FUNKTION

FUNKTION (*Parkhaus) init_parkhaus (Ganzzahl maximale_parkdauer, Ganzzahl anzahl_parkplaetze): //Die Variablen NICHT selber festlegen in main sondern von Simulationsparameter ablesen
    Parkhaus *p_parkhaus = malloc(sizeof(Parkhaus))
    p_parkhaus->maximale_parkdauer = maximale_parkdauer
    Parkplatz *p_parkplatz = malloc(sizeof(Parkplatz) * anzahl_parkplaetze)
    p_parkhaus->p_parkplaetze = p_parkplatz 
    p_parkhaus->p_erstes_kfz_in_der_warteschlange = NULL
    p_parkhaus->anzahl_parkplaetze = anzahl_parkplaetze
    p_parkhaus->belegte_parkplaetze = 0
    GIB p_parkhaus
ENDE FUNKTION 

FUNKTION (*Kfz) init_kfz (Parkhaus *p_parkhaus, Ganzzahl id):
    Kfz *p_kfz = malloc(sizeof(Kfz))
    p_kfz->id = id
    p_kfz->parkdauer = 0
    p_kfz->verbleibende_parkdauer = p_parkhaus->maximale_parkdauer //verbleibend = maximal da Kfz noch keine Zeit im Parkhaus verbracht hat
    p_kfz->p_naechstes_kfz = NULL //wird bei kfz_hinzufügen_warteschlange gesetzt
    GIB p_kfz
ENDE FUNKTION 

FUNKTION kfz_hinzufügen_warteschlange(Parkhaus *p_parkhaus, Kfz *p_kfz):
    WENN p_parkhaus->p_erstes_kfz_in_der_warteschlange == NULL:
        p_parkhaus->p_erstes_kfz_in_der_warteschlange = p_kfz
        p_kfz->p_naechstes_kfz = NULL
        GIB 0
        ENDE FUNKTION
    ENDE WENN
    Kfz *p_aktuelles_kfz = p_parkhaus->p_erstes_kfz_in_der_warteschlange
    SOLANGE (p_aktuelles_kfz->p_naechstes_kfz != NULL):
        p_aktuelles_kfz = p_aktuelles_kfz->p_naechstes_kfz
    ENDE SOLANGE
    p_aktuelles_kfz->p_naechstes_kfz = p_kfz
    p_kfz->p_naechstes_kfz = NULL
    ENDE FUNKTION

FUNKTION (*Kfz) entferne_kfz_warteschlange(Parkhaus *p_parkhaus):
    WENN p_parkhaus->p_erstes_kfz_in_der_warteschlange == NULL:
        Ausgabe: "Keine kfzs in der Warteschlange"
        GIB 0
        ENDE FUNKTION
    ENDE WENN
    WENN p_parkhaus->p_erstes_kfz_in_der_warteschlange->naechstes_kfz == NULL:
        Kfz *p_kfz_zwischenspeicher = p_parkhaus->p_erstes_kfz_in_der_warteschlange
        p_parkhaus->p_erstes_kfz_in_der_warteschlange = NULL
        GIB p_kfz_zwischenspeicher
        ENDE FUNKTION 
    ENDE WENN
    Kfz *p_kfz_zwischenspeicher = p_parkhaus->p_erstes_kfz_in_der_warteschlange
    p_parkhaus->p_erstes_kfz_in_der_warteschlange = p_kfz_zwischenspeicher->p_naechstes_kfz
    p_kfz_zwischenspeicher->p_naechstes_kfz = NULL
    GIB p_kfz_zwischenspeicher
ENDE FUNKTION

FUNKTION füge_kfz_hinzu(Parkhaus *p_parkhaus, Kfz *p_kfz):
    Ganzzahl freie_parkplaetze = p_parkhaus->anzahl_parkplaetze - p_parkhaus->belegte_parkplaetze

    min = 0
    max = frei_parkplaetze
    Ganzzahl zufaelliger_parkplatz = 0
    zufaelliger_parkplatz = min + (rand() % (max - min + 1))      
    FÜR Ganzzahl i = 0, i < p_parkhaus->anzahl_parkplaetze, i++:
        WENN !zufaelliger_parkplatz
            p_parkhaus->p_parkplaetze[i].belegt = 1
            p_parkhaus->p_parkplaetze[i].p_kfz = p_kfz
            p_parkhaus->belegte_parkplaetze = p_parkhaus->belegte_parkplaetze + 1
            GIB 0
            ENDE FUNKTION
        ENDE WENN    
        WENN !p_parkhaus->p_parkplaetze[i].belegt
            freie_parkplaetze = freie_parkplaetze - 1
        ENDE WENN    
        // Von allen freien Parkplätzen wird eins zufällig ausgesucht zum parken. Erst wenn die Variable "freie_parkplaetze" 0 erreicht wird das Kfz geparkt.
        // Es wird sozusagen simuliert an wie vielen freien Parkplätzen das Kfz vorbeWENNährt bevor es parkt
    ENDE FÜR
ENDE FUNKTION

FUNKTION enferne_kfz(Parkhaus *p_parkhaus, Kfz *p_kfz):
    FÜR Ganzzahl i = 0, i < p_parkhaus->anzahl_parkplaetze, i++:
        WENN p_kfz == p_parkhaus->p_parkplaetze[i].p_kfz:
            p_parkhaus->p_parkplaetze[i].belegt = 0
            free(p_kfz)
            p_parkhaus->p_parkplaetze[i].p_kfz = NULL
            p_parkhaus->belegte_parkplaetze = p_parkhaus->belegte_parkplaetze - 1
            GIB NULL
            ENDE FUNKTION
        ENDE WENN
    ENDE FÜR
    Ausgabe: "Dieses Kfz wurde im Parkhaus nicht gefunden" //Fehlermeldung
ENDE FUNKTION

FUNKTION enferne_kfzs_maximale_parkdauer(Parkhaus *p_parkhaus):
    FÜR Ganzahl i = 0, i < i < p_parkhaus->anzahl_parkplaetze, i++:
        WENN p_parkhaus->p_parkplaetze[i].p_kfz != NULL:
            WENN p_parkhaus->p_parkplaetze[i].p_kfz.parkdauer >= p_parkhaus->maximale_parkdauer:
                p_parkhaus->p_parkplaetze[i].belegt = 0
                free(p_parkhaus->p_parkplaetze[i]-p_kfz)
                p_parkhaus->p_parkplaetze[i]->Kfz = NULL //Falls Parkdauer überschritten fährt das Kfz raus
            ENDE WENN
        ENDE WENN
    ENDE FÜR
ENDE FUNKTION

FUNKTION simuliere_zeitabschnitt(Parkhaus *p_parkhaus, Simulationsparameter *p_simulationsparameter):
    //jedem Fahrzeug die aktuelle parkzeit um 1 erhöhen
    FÜR Ganzzahl i = 0, i < p_parkhaus->anzahl_parkplaetze, i++:
        WENN p_parkhaus->p_parkplaetze[i].belegt
            p_parkhaus->p_parkplaetze[i].p_kfz->parkdauer = p_parkhaus->p_parkplaetze[i].p_kfz->parkdauer + 1
        ENDE WENN
    ENDE FÜR    

    Ganzzahl min = 0
    Ganzzahl max = 1
    Fließkommazahl zufall = 0
    zufall = min + ((Fließkommazahl)rand() / RAND_MAX) * (max - min) // liefert eine zufällige Kommazahl zwischen 0-1 (wird benötigt bezüglich des Vergleichs mit der Wahrscheinlichkeit)
    //RAND_MAX ist eine C definierte Konstante die den maximalwert von rand liefert.
    enferne_kfzs_maximale_parkdauer(Parkhaus *p_parkhaus)
    WENN zufall <= p_simulationsparameter->wahrscheinlichkeit_neues_kfz: // ">=" statt ">" da falls Wahrscheinlichkeit = 1 und zufall = 1 soll trotzdem Kfz rein. Bei 100% wahrscheinlichkeit MUSS eine Kfz rein
        Kfz *p_neues_kfz = init_car(Parkhaus *p_parkhaus)
        kfz_hinzufügen_warteschlange(Parkhaus *p_parkhaus, Kfz *p_neues_kfz)
    ENDE WENN 

    zufall = min + ((Fließkommazahl)rand() / RAND_MAX) * (max - min)
    min = 0
    max = p_parkhaus->belegte_parkplatze - 1
    Ganzzahl rausfahrendes_kfz = min + rand() % (max - min + 1)
    
    WENN zufall <= p_simulationsparameter->wahrscheinlichkeit_kfz_rausfahren:
        enferne_kfz(Parkhaus *p_parkhaus, Kfz *p_parkhaus->p_parkplaetze[rausfahrendes_kfz]->kfz)
    ENDE WENN
    Ganzzahl freie_parkplaetze_wahrheitsvariable = 0; // 0 = keine freien Parkplaetze. 1 = freie Parkplaetze
    freie_parkplaetze_wahrheitsvariable = platz_garage(Parkhaus *p_parkhaus)

    WENN freie_parkplaetze_wahrheitsvariable AND p_parkhaus->erster_in_der_warteschlange != NULL:
        Kfz *p_kfz_hinzufügen_aus_queue = entferne_kfz_warteschlange(Parkhaus *p_parkhaus)
        füge_kfz_hinzu(Parkhaus *p_parkhaus, Kfz *p_kfz_hinzufügen_aus_queue)
    ENDE WENN
ENDE FUNKTION
