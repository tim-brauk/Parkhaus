FUNCTION init_parkhaus (Parkhaus *p_parkhaus, Ganzzahl parkplaetze, Ganzzahl maximale_parkdauer, Auto *p_erstes_auto_in_der_warteschlange):
    p_parkhaus->parkplaetze = parkplaetze
    p_parkhaus->maximale_parkdauer = maximale_parkdauer
    p_parkhaus->p_erstes_auto_in_der_warteschlange = p_erstes_auto_in_der_warteschlange
END FUNCTION

FUNCTION init_auto (Parkhaus *p_parkhaus, Auto *p_auto, Ganzzahl id, Auto p_naechstes_auto):
    p_auto.id = id
    p_auto.parkdauer = 0
    p_auto.restliche_parkdauer = p_parkhaus.maximale_parkdauer - aktuelle_parkdauer // Hier nochmal besprechen
    p_auto->p_naechstes_auto = p_naechstes_auto
END FUNCTION

FUNCTION add_car_queue(Parkhaus *p_parkhaus, Auto *p_auto):
    IF p_parkhaus->p_erstes_auto_in_der_warteschlange == NULL:
        p_parkhaus->erstes_auto_in_der_warteschlange = p_auto
        END FUNCTION //return
    END IF
    Ganzzahl *p_aktuelles_auto = malloc(sizeof(Auto))
    p_aktuelles_auto = p_parkhaus->erstes_auto_in_der_warteschlange

        p_aktuelles_auto = p_aktuelles_auto->naechstes_auto

END FUNCTION

    END FOR
    
    min = 0
    max = frei_parkplaetze
    Ganzzahl zufaelliger_parkplatz = 0

        END IF

FUNCTION simulate_timestep(parkhaus pparkhaus):
    Ganzzahl min = 0
    Ganzzahl max = 100
    Ganzzahl neue_autos = 0
    neue_autos = min + rand() % (max - min + 1) //zwischen 0 und 100 Autos die reinfahren wollen+

    FOR Ganzahl i = 0, i < (sizeof(pparkhaus.parkplaetze) / sizeof(parkplatz)), i++:
        IF pparkhaus.parkplaetze[i].A != NULL:
            IF pparkhaus.parkplaetze[i].A.parkdauer >= pparkhaus.maximale_parkdauer:
                pparkhaus.parkplaetze[i].belegt = 0
                pparkhaus.parkplaetze[i].A = NULL //Falls Parkdauer überschritten fährt das Auto raus
            END IF
        END IF
    END FOR

    FOR Ganzzahl i = 0, i < neue_autos, i++:
        IF pparkhaus.erstes_auto_in_der_warteschlange == NULL:
            A aauto = NULL
            init_auto(parkhaus pparkhaus, A aauto, Ganzzahl id, A NULL) //Besprechung bezüglich Logik
            pparkhaus.erstes_auto_in_der_warteschlange = aauto
        ELSE
            A aauto = NULL
            init_auto(parkhaus pparkhaus, A aauto, Ganzzahl id, A NULL)
            add_car_queue(parkhaus pparkhaus, A aauto)
        END IF
    END FOR

    Ganzzahl aktuell_autos_im_parkhaus = 0

    FOR Ganzahl i = 0, i < (sizeof(pparkhaus.parkplaetze) / sizeof(parkplatz)), i++:
        IF pparkhaus.parkplaetze[i].A != NULL:
            aktuell_autos_im_parkhaus = aktuell_autos_im_parkhaus + 1
        END IF
    END FOR
    
    Ganzzahl rausfahrende_autos = 0
    min = 0
    max = aktuell_autos_im_parkhaus
    rausfahrende_autos = min + rand() % (max - min + 1)

    FOR Ganzzahl i = 0, i < rausfahrende_autos, i++:
        IF pparkhaus.parkplaetze[i].belegt:
            pparkhaus.parkplaetze[i].beleg = 0
            pparkhaus.parkplaetze[i].A = NULL
        END IF
    END FOR

    Ganzzahl freie_parkplaetze = 0; // 0 = keine freien Parkplaetze. 1 = freie Parkplaetze

    FOR Ganzzahl i = 0, i < (sizeof(pparkhaus.parkplaetze) / sizeof(parkplatz)), i++:
        IF(pparkhaus.parkplaetze[i] == NULL):
            frei_parkplaetze = 1
        END IF
    END FOR //Überprüfen ob freie Parkplaetze vorhanden

    WHILE pparkhaus.erster_in_der_warteschlange != NULL AND frei_parkplaetze: //Solange freie Parkplätze und Autos in der Warteschlange werden autos ins parkhaus hinzugefügt
        add_car(parkhaus pparkhaus, A pparkhaus.erster_in_der_warteschlange)
        FOR Ganzzahl i = 0, i < (sizeof(pparkhaus.parkplaetze) / sizeof(parkplatz)), i++:
            IF(pparkhaus.parkplaetze[i] == NULL):
                frei_parkplaetze = 1
            END IF
        END FOR //Überprüfen ob freie Parkplaetze vorhanden
    END WHILE
END FUNCTION
