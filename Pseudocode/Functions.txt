FUNCTIONS

FUNCTION (Parkhaus) init_parkhaus (Parkplatz *p_parkplaetze, Ganzzahl maximale_parkdauer):
    Parkhaus *p_parkhaus = malloc(sizeof(Parkhaus))
    p_parkhaus->p_parkplaetze = p_parkplaetze
    p_parkhaus->maximale_parkdauer = maximale_parkdauer
    p_parkhaus->p_erstes_auto_in_der_warteschlange = NULL
END FUNCTION return p_parkhaus

FUNCTION (Auto) init_auto (Parkhaus *p_parkhaus, Ganzzahl id):
    Auto *p_auto = malloc(sizeof(Auto))
    p_auto.id = id
    p_auto.parkdauer = 0
    p_auto->p_naechstes_auto = NULL //wird bei add_car_queue gesetzt
END FUNCTION return p_auto

FUNCTION add_car_queue(Parkhaus *p_parkhaus, Auto *p_auto):
    IF p_parkhaus->p_erstes_auto_in_der_warteschlange == NULL:
        p_parkhaus->p_erstes_auto_in_der_warteschlange = p_auto
        p_auto->p_naechstes_auto = NULL
        END FUNCTION //return
    END IF
    Auto *p_aktuelles_auto = p_parkhaus->p_erstes_auto_in_der_warteschlange
    WHILE (p_aktuelles_auto->p_naechstes_auto != NULL):
        p_aktuelles_auto = p_aktuelles_auto->p_naechstes_auto
    END WHILE
    p_aktuelles_auto->p_naechstes_auto = p_auto
    p_auto->p_naechstes_auto = NULL
END FUNCTION

FUNCTION (Auto) remove_car_queue(Parkhaus *p_parkhaus):
    IF p_parkhaus->p_erstes_auto_in_der_warteschlange == NULL:
        Ausgabe: "Keine Autos in der Warteschlange"
        END FUNCTION return NULL
    END IF
    IF p_parkhaus->p_erstes_auto_in_der_warteschlange->naechstes_auto == NULL:
        Auto *p_auto_zwischenspeicher = p_parkhaus->p_erstes_auto_in_der_warteschlange
        p_parkhaus->p_erstes_auto_in_der_warteschlange = NULL
        END FUNCTION return p_auto_zwischenspeicher
    END IF
    Auto *p_auto_zwischenspeicher = malloc(sizeof(Auto))
    p_auto_zwischenspeicher = p_parkhaus->p_erstes_auto_in_der_warteschlange->p_naechstes_auto
    p_parkhaus->p_erstes_auto_in_der_warteschlange->p_naechstes_auto = NULL
    p_parkhaus->p_erstes_auto_in_der_warteschlange = p_auto_zwischenspeicher
END FUNCTION return p_auto_zwischenspeicher

FUNCTION add_car(Parkhaus *p_parkhaus, Auto *p_auto):
    Ganzzahl freie_parkplaetze = 0
    FOR Ganzzahl i = 0, i<(sizeof(p_parkhaus.parkplaetze) / sizeof(parkplaetz)), i++:
        IF !p_parkhaus.parkplaetze 
            freie_parkplaetze = freie_parkplaetze + 1
        END IF
    END FOR
    
    min = 0
    max = frei_parkplaetze
    Ganzzahl zufaelliger_parkplatz = 0
    zufaelliger_parkplatz = min + (rand() % (max - min + 1)) 

    FOR Ganzzahl i = 0, i < (sizeof(p_parkhaus.parkplaetze) / 4), i++: // Besprechung nötig wegen der Liste parkplaetze
        IF !zufaelliger_parkplatz
            p_parkhaus->parkplaetze[i].belegt = 1
            p_parkhaus->parkplaetze[i].auto = p_auto
            END FUNCTION
        END IF    
        
        IF !p_parkhaus->parkplaetze[i].belegt
            freie_parkplaetze = freie_parkplaetze - 1
        END IF    
    // Von allen freien Parkplätzen wird eins zufällig ausgesucht zum parken. Erst wenn die Variable "freie_parkplaetze" 0 erreicht wird das Auto geparkt.
    // Es wird sozusagen simuliert an wie vielen freien Parkplätzen das Auto vorbeifährt bevor es parkt
END FUNCTION
    

FUNCTION remove_car(Parkhaus *p_parkhaus, Auto *p_auto):
    FOR Ganzzahl i = 0, i < (sizeof(p_parkhaus->parkplaetze) / 4), i++: // Besprechung nötig wegen der Liste parkplaetze
        IF p_auto == p_parkhaus->parkplaetze[i].auto:
            p_parkhaus->parkplaetze[i].belegt = 0
            free(p_auto)
            p_parkhaus->parkplaetze[i].auto = NULL
            END FUNCTION
        END IF
    END FOR
    Ausgabe: "Dieses Auto wurde im Parkhaus nicht gefunden" //Fehlermeldung
END FUNCTION

FUNCTION remove_cars_max_parktime(Parkhaus *p_parkhaus):
    FOR Ganzahl i = 0, i < (sizeof(p_parkhaus->parkplaetze) / sizeof(parkplatz)), i++:
        IF p_parkhaus->parkplaetze[i].auto != NULL:
            IF p_parkhaus->parkplaetze[i].auto.parkdauer >= p_parkhaus->maximale_parkdauer:
                p_parkhaus->parkplaetze[i].belegt = 0
                free(p_parkhaus->parkplaetze[i]->auto)
                p_parkhaus->parkplaetze[i]->auto = NULL //Falls Parkdauer überschritten fährt das Auto raus
            END IF
        END IF
    END FOR
END FUNCTION

FUNCTION check_garage_for_space(Parkhaus *p_parkhaus): // 1 bei freien Parkplätzen. 0 wenn keine freien parkplätze
    FOR Ganzzahl i = 0, i < (sizeof(p_parkhaus->parkplaetze) / sizeof(parkplatz)), i++:
        IF(p_parkhaus->parkplaetze[i] == NULL):
            return 1
        END IF
    END FOR
    return 0
END FUNCTION
    

FUNCTION simulate_timestep(Parkhaus *p_parkhaus):
    //jedem Fahrzeug die aktuelle parkzeit um 1 erhöhen
    FOR Ganzzahl i = 0, i < sizeof(p_parkhaus->parkplaetze) / sizeof(Parkplatz), i++:
        IF p_parkhaus->parkplaetze[i].belegt
            p_parkhaus->parkplaetze[i]->auto.parkdauer = p_parkhaus->parkplaetze[i]->auto.parkdauer + 1
        END IF
    END FOR    
    
    Ganzzahl min = 0
    Ganzzahl max = 1
    Fließkommazahl zufall = 0
    zufall = min + ((Fließkommazahl)rand() / RAND_MAX) * (max - min) // liefert eine zufällige Kommazahl zwischen 0-1 (wird benötigt bezüglich des Vergleichs mit der Wahrscheinlichkeit)
    //RAND_MAX ist eine C definierte Konstante die den maximalwert von rand liefert.
    remove_cars_max_parktime(Parkhaus *p_parkhaus)
    IF zufall >= WAHRSCHEINLICHKEIT_NEUES_AUTO: // ">=" statt ">" da falls Wahrscheinlichkeit = 1 und zufall = 1 soll trotzdem Auto rein. Bei 100% wahrscheinlichkeit MUSS eine Auto rein
        Auto *p_neues_auto = init_car(Parkhaus *p_parkhaus)
        add_car_queue(Parkhaus *p_parkhaus, Auto *p_neues_auto)
    END IF

    Ganzzahl aktuell_autos_im_parkhaus = 0

    //Zählen wie viele Autos im Parkhaus
    FOR Ganzahl i = 0, i < (sizeof(p_parkhaus->parkplaetze) / sizeof(parkplatz)), i++:
        IF p_parkhaus->parkplaetze[i].auto != NULL:
            aktuell_autos_im_parkhaus = aktuell_autos_im_parkhaus + 1
        END IF
    END FOR

    zufall = min + ((Fließkommazahl)rand() / RAND_MAX) * (max - min)
    min = 0
    max = aktuell_autos_im_parkhaus - 1
    Ganzzahl rausfahrendes_auto = min + rand() % (max - min + 1)
    
    IF zufall >= WAHRSCHEINLICHKEIT_AUTO_RAUSFAHREN:
        remove_car(Parkhaus *p_parkhaus, Auto *p_parkhaus->parkplaetze[rausfahrendes_auto]->auto)
    END IF

    Ganzzahl freie_parkplaetze_wahrheitsvariable = 0; // 0 = keine freien Parkplaetze. 1 = freie Parkplaetze

    freie_parkplaetze_wahrheitsvariable = check_garage_for_space(Parkhaus *p_parkhaus)
    
    IF freie_parkplaetze_wahrheits AND p_parkhaus->erster_in_der_warteschlange != NULL:
        Auto *p_auto_hinzufügen_aus_queue = remove_car_queue(Parkhaus *p_parkhaus)
        add_car(Parkhaus *p_parkhaus, Auto *p_auto_hinzufügen_aus_queue)
    END IF
END FUNCTION
